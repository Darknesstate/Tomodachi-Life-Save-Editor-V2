name: Build Tomodachi V2
on: [push, workflow_dispatch]
jobs:
  build:
    runs-on: ubuntu-latest
    container: {image: devkitpro/devkitarm:latest}
    steps:
      - uses: actions/checkout@v4
      - name: Build
        run: |
          make
          mkdir -p build
          wget -q https://github.com/3DSGuy/Project_CTR/releases/download/makerom-v0.19.0/makerom-v0.19.0-ubuntu_x86_64.zip -O /tmp/m.zip
          python3 -m zipfile -e /tmp/m.zip /tmp/m
          cp /tmp/m/makerom /usr/local/bin/makerom
          chmod +x /usr/local/bin/makerom
          curl -fL https://github.com/Epicpkmn11/bannertool/releases/download/v1.2.2/bannertool.zip -o /tmp/b.zip
          python3 -m zipfile -e /tmp/b.zip /tmp/b
          cp /tmp/b/linux-x86_64/bannertool /usr/local/bin/bannertool
          chmod +x /usr/local/bin/bannertool
          BANNER=$(find . -maxdepth 2 -name banner.png | head -n 1)
          ICON=$(find . -maxdepth 2 -name icon.png | head -n 1)
          python3 -c "import wave,struct; w=wave.open('build/silent.wav','wb'); w.setnchannels(1); w.setsampwidth(2); w.setframerate(22050); w.writeframes(struct.pack('<h',0)*22050); w.close()"
          bannertool makebanner -i "$BANNER" -a build/silent.wav -o build/banner.bnr
          bannertool makesmdh -s "Tomodachi Life Save Editor V2" -l "Tomodachi Life Save Editor V2" -p "Darkness Metal" -i "$ICON" -o build/icon.icn
          ELF=$(find . -maxdepth 3 -name "*.elf" | head -n 1)
          makerom -f cia -o TomodachiLifeSaveEditorV2.cia -DAPP_ENCRYPTED=false -rsf TomodachiReal.rsf -target t -exefslogo -elf "$ELF" -icon build/icon.icn -banner build/banner.bnr -major 1 -minor 0 -micro 0
      - name: Upload artefato
        uses: actions/upload-artifact@v4
        if: always()
        with:
          name: TomodachiLifeSaveEditorV2
          path: |
            *.3dsx
            *.smdh
            *.cia
